        const NOTES = ['C', 'C#', 'D', 'D#', 'E', 'F', 'F#', 'G', 'G#', 'A', 'A#

        const normalizeTone = (tone) => {
            if (!tone) return 'C';
            const flats = { 'Db': 'C#', 'Eb': 'D#', 'Gb': 'F#', 'Ab': 'G#', 'Bb'
            for (let [f, s] of Object.entries(flats)) {
                if (tone.startsWith(f)) return s + tone.slice(2);
            }
            return tone;
        };

        const transposeChord = (chord, semitones) => {
            if (!chord) return "";
            let normalized = normalizeTone(chord);
            const rootMatch = normalized.match(/^([A-G]#?)/);
            if (!rootMatch) return chord;

            const root = rootMatch[1];
            const suffix = normalized.slice(root.length);
            const currentIndex = NOTES.indexOf(root);
            const newIndex = (currentIndex + semitones + 12) % 12;
            return NOTES[newIndex] + suffix;
        };

        const stripChords = (text) => {
            // Elimina líneas que parecen solo acordes o fragmentos de acordes
            return text.split('\n').filter(line => {
                return !line.match(/^[A-G]#?[m7maj|sus|add|5]*\s*$/);
            }).join('\n');
        };

        const normalizeText = (text) => {
            return text.normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLower
        };

        async function enterStageMode() {
            const eventSongs = await db.eventSongs.where('eventId').equals(currentEventId).sortBy('order').toArray();
            currentStageSongs = [];
            for (let es of eventSongs) {
                currentStageSongs.push(await db.songs.get(es.songId));

            renderStage();
            showView('stage');

        function renderStage() {
            const contentDiv = document.getElementById('stage-content');
            let html = "";

            currentStageSongs.forEach(song => {
                html += `<h1 style="font-family:sans-serif; text-align:center;">${song.title}</h1>`;

                const lines = song.content.split('\n');
                lines.forEach(line => {
                    if (line.startsWith('[')) {
                        html += `<span class="section-title">${line.substring(1,
                    } else {
                        // Transponer cada acorde en la línea
                        const transposedLine = line.replace(/([A-G]#?[m7maj|sus|add|5]*)/g, (match) => {
                            return transposeChord(match, currentTranspose);
                        });
                        html += transposedLine + "\n";
                    }
                });
                html += "<br><hr><br>";
            });
            contentDiv.innerHTML = html;
        }

                document.getElementById('btn-scroll').innerText = "Scroll: OFF";

                scrollInterval = setInterval(() => {
                    window.scrollBy(0, 1);
                }, 50);
            }
        }

        function exitStageMode() {
            if (scrollInterval) toggleScroll();
